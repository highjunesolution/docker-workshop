// ดู logs แบบ realtime
docker logs -f my-nginx
-f = follow

คล้าย tail -f
ถ้ามี request เข้ามา nginx log จะโผล่ต่อเนื่อง
ออกด้วย ctrl + c

// เข้าไปข้างใน Container (linux container)****
docker exec -it my-nginx bash

docker exec = สั่ง command ภายใน container ที่กำลัง running
-i = interactive เปิด stdin ไว้ให้เราพิมพ์
-t = pseudo-TTY สร้าง terminal ให้เรา

ดังนั้น
docker exec -it my-nginx bash
อ่านเป็น เปิด bash แบบ interactive terminal ภายใน my-nginx

// container ไม่จำเป็นต้องมี bash
บาง image มี bash
บาง image ไม่มี 
โดยเฉพาะพวก image เล็กๆ เช่น alpine

docker exec -it container bash
docker exec -it container sh
